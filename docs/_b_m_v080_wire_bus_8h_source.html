60 wireBusError_t readRegister(uint16_t regAddr, uint16_t *data, uint16_t length, size_t &bytesRead)
61 {
62 if (_wire == nullptr || data == nullptr)
63 return wireBusFail;
64
65 bytesRead = 0;
66
67 // Register address is already shifted by callback (word address -> byte address)
68 // The shifted value represents the byte address, send as MSB, LSB
69 uint8_t regAddrMsb = (uint8_t)((regAddr >> 8) & 0xFF);
70 uint8_t regAddrLsb = (uint8_t)(regAddr & 0xFF);
71
72 // Write register address (with stop condition)
73 _wire->beginTransmission(_address);
74 _wire->write(regAddrMsb);
75 _wire->write(regAddrLsb);
76 if (_wire->endTransmission() != 0)
77 return wireBusFail;
78
79 // Request data (each register is 2 bytes, so length * 2 bytes total)
80 uint8_t bytesToRead = length * 2;
81 uint8_t bytesReceived = _wire->requestFrom(_address, bytesToRead);
82
83 if (bytesReceived != bytesToRead)
84 return wireBusFail;
85
86 // Read data (MSB first, as per BMV080 spec)
87 for (uint16_t i = 0; i < length; i++)
88 {
89 if (_wire->available() < 2)
90 return wireBusFail;
91 uint8_t msb = _wire->read();
92 uint8_t lsb = _wire->read();
93 data[i] = (uint16_t)((msb << 8) | lsb);
94 bytesRead += 2;
95 }
96
97 return wireBusOK;
98 }
99
107 wireBusError_t writeRegister(uint16_t regAddr, const uint16_t *data, uint16_t length)
108 {
109 if (_wire == nullptr || data == nullptr)
110 return wireBusFail;
111
112 // Register address is already shifted by callback (word address -> byte address)
113 // The shifted value represents the byte address, send as MSB, LSB
114 uint8_t regAddrMsb = (uint8_t)((regAddr >> 8) & 0xFF);
115 uint8_t regAddrLsb = (uint8_t)(regAddr & 0xFF);
116
117 // Start transmission and write register address (MSB first)
118 _wire->beginTransmission(_address);
119 _wire->write(regAddrMsb);
120 _wire->write(regAddrLsb);
121
122 // Write data (MSB first, as per BMV080 spec)
123 for (uint16_t i = 0; i < length; i++)
124 {
125 _wire->write((uint8_t)((data[i] >> 8) & 0xFF)); // MSB
126 _wire->write((uint8_t)(data[i] & 0xFF)); // LSB
127 }
128
129 uint8_t txError = _wire->endTransmission();
130 if (txError != 0)
131 return wireBusFail;
132
133 return wireBusOK;
134 }
