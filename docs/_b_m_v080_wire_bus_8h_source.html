58 wireBusError_t readRegister(uint16_t regAddr, uint16_t *data, uint16_t length, size_t &bytesRead)
59 {
60 if (_wire == nullptr || data == nullptr)
61 return wireBusFail;
62
63 bytesRead = 0;
64
65 // Register address is already shifted by callback (word address -> byte address)
66 // The shifted value represents the byte address, send as MSB, LSB
67 uint8_t regAddrMsb = (uint8_t)((regAddr >> 8) & 0xFF);
68 uint8_t regAddrLsb = (uint8_t)(regAddr & 0xFF);
69
70 // Write register address (with stop condition)
71 _wire->beginTransmission(_address);
72 _wire->write(regAddrMsb);
73 _wire->write(regAddrLsb);
74 if (_wire->endTransmission() != 0)
75 return wireBusFail;
76
77 // Request data (each register is 2 bytes, so length * 2 bytes total)
78 uint8_t bytesToRead = length * 2;
79 uint8_t bytesReceived = _wire->requestFrom(_address, bytesToRead);
80
81 if (bytesReceived != bytesToRead)
82 return wireBusFail;
83
84 // Read data (MSB first, as per BMV080 spec)
85 for (uint16_t i = 0; i < length; i++)
86 {
87 if (_wire->available() < 2)
88 return wireBusFail;
89 uint8_t msb = _wire->read();
90 uint8_t lsb = _wire->read();
91 data[i] = (uint16_t)((msb << 8) | lsb);
92 bytesRead += 2;
93 }
94
95 return wireBusOK;
96 }

105 wireBusError_t writeRegister(uint16_t regAddr, const uint16_t *data, uint16_t length)
106 {
107 if (_wire == nullptr || data == nullptr)
108 return wireBusFail;
109
110 // Register address is already shifted by callback (word address -> byte address)
111 // The shifted value represents the byte address, send as MSB, LSB
112 uint8_t regAddrMsb = (uint8_t)((regAddr >> 8) & 0xFF);
113 uint8_t regAddrLsb = (uint8_t)(regAddr & 0xFF);
114
115 // Start transmission and write register address (MSB first)
116 _wire->beginTransmission(_address);
117 _wire->write(regAddrMsb);
118 _wire->write(regAddrLsb);
119
120 // Write data (MSB first, as per BMV080 spec)
121 for (uint16_t i = 0; i < length; i++)
122 {
123 _wire->write((uint8_t)((data[i] >> 8) & 0xFF)); // MSB
124 _wire->write((uint8_t)(data[i] & 0xFF)); // LSB
125 }
126
127 uint8_t txError = _wire->endTransmission();
128 if (txError != 0)
129 return wireBusFail;
130
131 return wireBusOK;
132 }
